## 3. Target Relational SQL Schemas (DDL)

Below is the complete SQL DDL specification for creating all 9 tables required for ATEM Field on Supabase.

```sql
-- Enable UUID extension if needed
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- ==========================================
-- 1. TECHNICIANS (Técnicos)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.technicians (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL,
    phone TEXT NOT NULL DEFAULT 'Pendente (Cadastrar via Telegram)',
    telegram_id BIGINT UNIQUE,
    email TEXT DEFAULT 'Pendente',
    role TEXT NOT NULL DEFAULT 'Técnico de Campo',
    avatar_url TEXT DEFAULT 'https://images.unsplash.com/photo-1535713875002-d1d0cf377fde?auto=format&fit=crop&w=250&q=80',
    status TEXT NOT NULL DEFAULT 'Ativo' CHECK (status IN ('Ativo', 'Inativo', 'Férias')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 2. EQUIPMENT CATEGORIES (Categorias de Equipamentos)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.equipment_categories (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    name TEXT NOT NULL UNIQUE,
    description TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 3. TEST TEMPLATES (Templates de Testes/Checklist)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.test_templates (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    category_id BIGINT NOT NULL REFERENCES public.equipment_categories(id) ON DELETE CASCADE,
    analyzer_type TEXT NOT NULL,
    required_parameters JSONB NOT NULL DEFAULT '[]'::jsonb,
    is_mandatory INTEGER NOT NULL DEFAULT 1,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 4. ANALYZERS (Analisadores e Padrões de Teste)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.analyzers (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    asset_tag TEXT NOT NULL UNIQUE,
    name TEXT NOT NULL,
    type TEXT NOT NULL,
    serial_number TEXT NOT NULL,
    brand_model TEXT NOT NULL,
    status TEXT NOT NULL DEFAULT 'Disponível' CHECK (status IN ('Disponível', 'Reservado', 'Em Uso', 'Em Manutenção')),
    calibration_date DATE,
    calibration_expiry DATE,
    current_location TEXT NOT NULL DEFAULT 'Almoxarifado Central',
    notes TEXT DEFAULT '',
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 5. APPOINTMENTS (Agenda de Atendimentos)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.appointments (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    external_id TEXT UNIQUE,
    title TEXT NOT NULL,
    client_name TEXT NOT NULL,
    client_address TEXT,
    client_contact TEXT DEFAULT 'Sincronizado da Agenda Online',
    technician_id BIGINT REFERENCES public.technicians(id) ON DELETE SET NULL,
    technician_names TEXT DEFAULT 'Equipe de Campo',
    appointment_date DATE NOT NULL,
    appointment_time TEXT DEFAULT '08:00',
    status TEXT NOT NULL DEFAULT 'Agendado' CHECK (status IN ('Agendado', 'Em Andamento', 'Concluído', 'Cancelado')),
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 6. APPOINTMENT_TECHNICIANS (Relacionamento N:N Atendimentos <-> Técnicos)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.appointment_technicians (
    appointment_id BIGINT NOT NULL REFERENCES public.appointments(id) ON DELETE CASCADE,
    technician_id BIGINT NOT NULL REFERENCES public.technicians(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now()),
    PRIMARY KEY (appointment_id, technician_id)
);

-- ==========================================
-- 7. APPOINTMENT_EQUIPMENTS (Equipamentos do Hospital Atendido)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.appointment_equipments (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    appointment_id BIGINT NOT NULL REFERENCES public.appointments(id) ON DELETE CASCADE,
    category_id BIGINT REFERENCES public.equipment_categories(id) ON DELETE SET NULL,
    model TEXT,
    serial_number TEXT,
    tag TEXT,
    sector TEXT DEFAULT 'Assistência Técnica',
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 8. RESERVATIONS (Reservas e Checkout/Checkin de Analisadores)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.reservations (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    appointment_id BIGINT NOT NULL REFERENCES public.appointments(id) ON DELETE CASCADE,
    analyzer_id BIGINT NOT NULL REFERENCES public.analyzers(id) ON DELETE CASCADE,
    technician_id BIGINT NOT NULL REFERENCES public.technicians(id) ON DELETE CASCADE,
    reserved_date DATE NOT NULL,
    status TEXT NOT NULL DEFAULT 'Reservado' CHECK (status IN ('Reservado', 'Em Uso', 'Devolvido', 'Cancelado')),
    checkout_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()),
    checkin_at TIMESTAMPTZ,
    notes TEXT DEFAULT 'Reserva efetuada via sistema/Telegram',
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);

-- ==========================================
-- 9. NOTIFICATION_LOGS (Logs de Notificações Telegram/Cron)
-- ==========================================
CREATE TABLE IF NOT EXISTS public.notification_logs (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    technician_id BIGINT NOT NULL REFERENCES public.technicians(id) ON DELETE CASCADE,
    appointment_id BIGINT REFERENCES public.appointments(id) ON DELETE SET NULL,
    type TEXT NOT NULL,
    message TEXT NOT NULL,
    sent_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc'::text, now())
);
```

### Row-Level Security (RLS) & Public Grants
To comply with Supabase domain security guidelines while maintaining compatibility with dev mode:

```sql
-- Enable RLS on all tables
ALTER TABLE public.technicians ENABLE ROW LEVEL SECURITY;
```

## Edge Cases
| # | Feature | Input | Observed Behavior |
|---|---------|-------|-------------------|
| 1 | Reservation Conflict | Attempting to reserve an analyzer (`PAT-104050`) already in state 'Reservado' or 'Em Uso' on the same date for a different appointment. | Returns HTTP 400 with message: `"O Patrimônio PAT-104050 (...) já está reservado/em uso para esta data."` |
| 2 | Duplicate Asset Tag | Creating a new analyzer with an existing `asset_tag` (e.g. `PAT-104050`). | Returns HTTP 400 with message: `"Já existe um analisador cadastrado com o Patrimônio PAT-104050."` |
| 3 | Multiple Technicians Parsing | Calendar title with raw string like `"Técnico: Christopher; Kauan (Duster)"`. | Regex cleans vehicle tags, splits by `;`, matching or auto-creating technicians and linking via `appointment_technicians`. |
| 4 | JSONB Parameter Parsing | `test_templates.required_parameters` passed as JSON string or string array. | Express route checks if string or array, serializing/deserializing appropriately for `jsonb` column storage. |
| 5 | Checkout/Checkin Flow | Calling `/checkout` sets `reservation.status` to 'Em Uso' and updates `analyzer.status` to 'Em Uso'. Calling `/checkin` sets `reservation.status` to 'Devolvido' and restores `analyzer.status` to 'Disponível'. | Analyzer status dynamically toggles between 'Disponível', 'Reservado', 'Em Uso', ensuring proper inventory tracking. |
